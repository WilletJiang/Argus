cmake_minimum_required(VERSION 3.20)
project(ah VERSION 0.2.1 LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
option(AH_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(AH_FUZZ "Build libFuzzer targets (Clang)" OFF)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(CURL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
if(AH_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(ah_json STATIC vendor/yyjson/yyjson.c)
target_include_directories(ah_json PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/vendor/yyjson>)
set_target_properties(ah_json PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(ah STATIC
  src/common.c
  src/core/context.c src/core/engine.c src/core/execute.c src/core/mailbox.c
  src/core/operations.c src/core/parallel.c src/core/recovery.c src/core/response.c
  src/core/results.c src/core/runtime.c src/core/session.c src/core/store.c
  src/providers/responses.c src/providers/stream.c
  src/tools/exec.c src/tools/file.c src/tools/file_read.c src/tools/local.c src/tools/schema.c)
add_library(ah::ah ALIAS ah)
target_include_directories(ah PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(ah PRIVATE src)
target_compile_definitions(ah PRIVATE _POSIX_C_SOURCE=200809L _GNU_SOURCE=1 _DARWIN_C_SOURCE=1)
target_compile_options(ah PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(ah PRIVATE ah_json CURL::libcurl SQLite::SQLite3 Threads::Threads)
if(UNIX AND NOT APPLE)
  target_link_libraries(ah PRIVATE m)
endif()
add_executable(ah_cli src/cli/main.c src/cli/output.c src/cli/rpc.c src/cli/credential.c src/cli/signals.c)
set_target_properties(ah_cli PROPERTIES OUTPUT_NAME ah)
target_include_directories(ah_cli PRIVATE src)
target_compile_definitions(ah_cli PRIVATE _POSIX_C_SOURCE=200809L _GNU_SOURCE=1 _DARWIN_C_SOURCE=1)
target_compile_options(ah_cli PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(ah_cli PRIVATE ah ah_json Threads::Threads)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  foreach(test core stream tools alloc parallel context results recovery cache)
      add_executable(test_${test} tests/test_${test}.c)
      target_include_directories(test_${test} PRIVATE src src/providers vendor/yyjson)
      target_compile_definitions(test_${test} PRIVATE _POSIX_C_SOURCE=200809L _GNU_SOURCE=1 _DARWIN_C_SOURCE=1)
      target_compile_options(test_${test} PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
      target_link_libraries(test_${test} PRIVATE ah ah_json Threads::Threads SQLite::SQLite3)
      add_test(NAME ${test} COMMAND test_${test})
      set_tests_properties(${test} PROPERTIES TIMEOUT 45)
  endforeach()
  target_sources(test_tools PRIVATE tests/file_cases.c)
  target_sources(test_parallel PRIVATE tests/parallel_recovery_cases.c)
endif()
install(TARGETS ah ah_json ah_cli EXPORT ahTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT ahTargets NAMESPACE ah:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ah)

if(BUILD_TESTING)
  add_executable(http_fixture tests/http_fixture.c)
  target_link_libraries(http_fixture PRIVATE ah)
  add_test(NAME http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_http.py $<TARGET_FILE:http_fixture>)
  set_tests_properties(http PROPERTIES TIMEOUT 60)
endif()
if(BUILD_TESTING)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:ah_cli>)
  set_tests_properties(cli PROPERTIES TIMEOUT 60)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/ahConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/ahConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ah)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/ahConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ahConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/ahConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ah)
add_executable(ah_embed examples/embed.c)
target_link_libraries(ah_embed PRIVATE ah)
if(BUILD_TESTING)
  add_test(NAME embed COMMAND ah_embed)
endif()

  target_include_directories(fuzz_stream PRIVATE include src src/providers vendor/yyjson)
  target_compile_definitions(fuzz_stream PRIVATE _POSIX_C_SOURCE=200809L _DARWIN_C_SOURCE=1)
  target_compile_options(fuzz_stream PRIVATE -fsanitize=fuzzer,undefined)
